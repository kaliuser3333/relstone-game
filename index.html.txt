<body>
    <div class="header">
        <div>Баланс</div>
        <div class="balance" id="balance">$1000</div>
    </div>

    <div class="game-container" id="grid">
        <!-- Тут будут клетки -->
    </div>

    <div class="controls">
        <button class="btn" id="action-btn" onclick="handleAction()">Начать игру (Ставка $10)</button>
    </div>

    <script>
        // Инициализация Telegram
        const tg = window.Telegram.WebApp;
        tg.ready();
        tg.expand();

        // Данные пользователя
        const user = tg.initDataUnsafe?.user;
        if (user) {
            console.log("User:", user.first_name);
        }

        // Состояние игры
        let balance = 1000;
        let gameActive = false;
        let minesCount = 3;
        let mines = [];
        let revealed = [];
        let betAmount = 10;

        // DOM элементы
        const balanceEl = document.getElementById('balance');
        const gridEl = document.getElementById('grid');
        const actionBtn = document.getElementById('action-btn');

        // Функции
        function updateBalance() {
            balanceEl.textContent = '$' + balance;
        }

        function createGrid() {
            gridEl.innerHTML = '';
            for (let i = 0; i < 25; i++) {
                const tile = document.createElement('div');
                tile.className = 'tile';
                tile.dataset.index = i;
                tile.onclick = () => handleTileClick(i);
                gridEl.appendChild(tile);
            }
        }

        function startGame() {
            if (balance < betAmount) {
                tg.showAlert("Недостаточно средств!");
                return;
            }
            balance -= betAmount;
            updateBalance();
            
            gameActive = true;
            revealed = [];
            mines = [];
            // Генерируем мины
            while (mines.length < minesCount) {
                const r = Math.floor(Math.random() * 25);
                if (!mines.includes(r)) mines.push(r);
            }
            
            createGrid();
            actionBtn.textContent = "Забрать выигрыш ($" + betAmount + ")";
            actionBtn.disabled = false;
        }

        function handleTileClick(index) {
            if (!gameActive || revealed.includes(index)) return;
            
            revealed.push(index);
            const tile = document.querySelector(`.tile[data-index="${index}"]`);
            
            if (mines.includes(index)) {
                // Проигрыш
                tile.classList.add('mine');
                tile.textContent = '💣';
                gameActive = false;
                actionBtn.disabled = true;
                actionBtn.textContent = "Игра окончена";
                tg.HapticFeedback.impactOccurred('heavy');
            } else {
                // Выигрыш
                tile.classList.add('diamond');
                tile.textContent = '💎';
                // Увеличиваем потенциальный выигрыш
                betAmount = Math.floor(betAmount * 1.2);
                actionBtn.textContent = "Забрать выигрыш ($" + betAmount + ")";
                tg.HapticFeedback.impactOccurred('light');
            }
        }

        function handleAction() {
            if (!gameActive) {
                startGame();
            } else {
                // Забрать выигрыш
                balance += betAmount;
                updateBalance();
                gameActive = false;
                actionBtn.textContent = "Начать игру (Ставка $10)";
                betAmount = 10;
                tg.showAlert(`Вы забрали $${betAmount - 10}!`); // Чистый выигрыш
            }
        }

        // Инициализация
        createGrid();
    </script>
